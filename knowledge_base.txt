إليك كود احترافي مبني على مكتبة python-telegram-bot (الإصدار الأخير 20+)، قمت بدمج System Instruction فيه ليتقمص الذكاء الاصطناعي دور موظف الشركة:

Python
import logging
from telegram import Update
from telegram.ext import ApplicationBuilder, ContextTypes, MessageHandler, filters
import google.generativeai as genai

# 1. إعداد Gemini (ضع مفتاحك هنا)
genai.configure(api_key="YOUR_GEMINI_API_KEY")
model = genai.GenerativeModel('gemini-1.5-flash')

# 2. تعليمات النظام (هذا ما سيجعل البوت خبير بـ ال تي)
SYSTEM_PROMPT = """
أنت مساعد ذكي لخدمة عملاء شركة ال تي (LT) للجوالات.
- أجب بلهجة محترمة وودودة.
- إذا سألك العميل عن مواصفات، قدمها بشكل منظم.
- إذا لم تكن متأكداً من معلومة، اطلب من العميل زيارة أقرب فرع أو الاتصال بالدعم الفني.
"""

async def respond(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_message = update.message.text
    
    # دمج تعليمات النظام مع سؤال المستخدم
    full_prompt = f"{SYSTEM_PROMPT}\n\nسؤال العميل: {user_message}"
    
    try:
        response = model.generate_content(full_prompt)
        await update.message.reply_text(response.text)
    except Exception as e:
        await update.message.reply_text("عذراً، حدث خطأ فني. حاول مرة أخرى.")

if __name__ == '__main__':
    # 3. إعداد تيليجرام (ضع التوكن الذي حصلت عليه من BotFather)
    app = ApplicationBuilder().token("YOUR_TELEGRAM_TOKEN").build()
    
    app.add_handler(MessageHandler(filters.TEXT & (~filters.COMMAND), respond))
    
    print("البوت يعمل الآن على تيليجرام...")
    app.run_polling()
